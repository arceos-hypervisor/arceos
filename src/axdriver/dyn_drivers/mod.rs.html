18/// Sets up the device driver subsystem.
19pub fn setup(dtb: usize) {
20    if dtb == 0 {
21        warn!("Device tree base address is 0, skipping device driver setup.");
22        return;
23    }
24    cache::setup_dma_api();
25    let dtb_virt = phys_to_virt(dtb.into());
26    if let Some(dtb) = NonNull::new(dtb_virt.as_mut_ptr()) {
27        rdrive::init(rdrive::Platform::Fdt { addr: dtb }).unwrap();
28        rdrive::register_append(&driver_registers());
29        // rdrive::probe_pre_kernel().unwrap();
30    }
31}

33#[allow(dead_code)]
34/// maps a mmio physical address to a virtual address.
35fn iomap(addr: PhysAddr, size: usize) -> Result<NonNull<u8>, OnProbeError> {
36    axklib::mem::iomap(addr, size)
37        .map_err(|e| match e {
38            AxError::NoMemory => OnProbeError::KError(rdrive::KError::NoMem),
39            _ => OnProbeError::Other(alloc::format!("{e:?}").into()),
40        })
41        .map(|v| unsafe { NonNull::new_unchecked(v.as_mut_ptr()) })
42}

44#[allow(dead_code)]
45fn parse_fdt_irq(intc: DeviceId, irq: &[u32]) -> IrqConfig {
46    let intc = rdrive::get::<rdif_intc::Intc>(intc).expect("No interrupt controller found");
47    let intc = intc.lock().unwrap();
48    let fdt_parse = intc.parse_dtb_fn().expect("No DTB parse function found");
49    fdt_parse(irq).unwrap()
50}

52fn driver_registers() -> impl Deref<Target = [DriverRegister]> {
53    unsafe extern "C" {
54        fn __sdriver_register();
55        fn __edriver_register();
56    }
57
58    unsafe {
59        let len = __edriver_register as usize - __sdriver_register as usize;
60
61        if len == 0 {
62            return DriverRegisterSlice::empty();
63        }
64
65        let data = core::slice::from_raw_parts(__sdriver_register as _, len);
66
67        DriverRegisterSlice::from_raw(data)
68    }
69}
